#!/bin/bash

# 🎬 SCRIPTUREMON - COMANDO GLOBAL ULTIMATE
# NOVA VERSÃO: Por padrão ativa ULTIMATE SYMBIOTIC (obra-prima)
# Backward compatibility mantida para versões antigas

# Cores majestosas
GOLD='\033[1;33m'
PURPLE='\033[0;35m'
CYAN='\033[0;36m'
GREEN='\033[0;32m'
NC='\033[0m'

# Diretório do Scripturemon
SCRIPTUREMON_HOME="/Users/clubproducoes/Digimundo/digimons/scripturemon"

# Banner Ultimate
echo ""
echo -e "${GOLD}✨ SCRIPTUREMON ULTIMATE SYMBIOTIC GLOBAL${NC}"
echo -e "${PURPLE}   Obra-prima com todos os conceitos recuperados${NC}"
echo ""

# Verificar se Ultimate está disponível
if [ -f "$SCRIPTUREMON_HOME/bin/scripturemon" ] && [ -f "$SCRIPTUREMON_HOME/SCRIPTUREMON_ULTIMATE_SYMBIOTIC.py" ]; then
    
    # DECISÃO: Ativar sempre ULTIMATE por padrão
    echo -e "${GOLD}🚀 Redirecionando para ULTIMATE SYMBIOTIC...${NC}"
    echo ""
    
    # Executar comando ultimate local
    exec "$SCRIPTUREMON_HOME/bin/scripturemon" "$@"
    
else
    echo -e "${CYAN}⚠️ Sistema Ultimate não configurado${NC}"
    echo ""
    echo -e "${GREEN}SETUP RÁPIDO:${NC}"
    echo -e "  cd $SCRIPTUREMON_HOME"
    echo -e "  ./ACTIVATE_ULTIMATE_SYMBIOTIC.sh"
    echo ""
    echo -e "${GREEN}FALLBACK DISPONÍVEL:${NC}"
    
    # Tentar versões antigas como fallback
    if command -v ollama &> /dev/null; then
        models=$(ollama list 2>/dev/null | grep scripturemon | head -1 | awk '{print $1}')
        if [ ! -z "$models" ]; then
            echo -e "  ${CYAN}Executando: ollama run $models${NC}"
            ollama run "$models"
        else
            echo -e "  ${CYAN}Nenhum modelo Scripturemon encontrado${NC}"
            echo -e "  Instale com: ollama pull mistral:latest"
        fi
    else
        echo -e "${RED}❌ Ollama não instalado${NC}"
        echo -e "   Instale: curl -fsSL https://ollama.com/install.sh | sh"
        exit 1
    fi
fi
